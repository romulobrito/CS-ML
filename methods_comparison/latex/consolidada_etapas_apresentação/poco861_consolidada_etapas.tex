\begin{frame}{Motiva\c{c}\~ao: transformar dados incompletos em uma estimativa de \Vp{}}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Problema cient\'ifico}
      \scriptsize
      Em carbonatos, a velocidade n\~ao depende apenas da porosidade total.
      Geometria e conectividade dos poros, mineralogia, fluido, press\~ao e
      unidade de fluxo tamb\'em controlam a resposta el\'astica.
    \end{block}
    \begin{block}{Problema operacional}
      \scriptsize
      \begin{itemize}
        \item Perfis cobrem o po\c{c}o, mas n\~ao resolvem toda a textura.
        \item Laborat\'orio e microCT trazem detalhe, mas em poucos pontos.
        \item O s\^onico mede \Vp{} continuamente, por\'em n\~ao explica
              sozinho a origem f\'isica das varia\c{c}\~oes.
      \end{itemize}
    \end{block}
    \column{0.50\textwidth}
    \begin{exampleblock}{Hip\'otese de trabalho}
      \scriptsize
      Nenhuma fonte isolada \'e suficiente. A solu\c{c}\~ao deve combinar:
      \begin{enumerate}
        \item cobertura espacial dos perfis;
        \item restri\c{c}\~oes petrof\'isicas e el\'asticas dos plugs;
        \item um modelo f\'isico interpret\'avel;
        \item corre\c{c}\~ao estat\'istica somente do erro remanescente.
      \end{enumerate}
    \end{exampleblock}
    \vspace{0.06cm}
    \begin{block}{Objetivo}
      \scriptsize
      Construir uma trilha de \Vp{} que preserve a interpreta\c{c}\~ao
      f\'isica, explicite as principais fontes de incerteza e limita\c{c}\~oes
      e evite que o ML apenas copie o perfil s\^onico.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Etapa 1: separar sinal de perfil de informa\c{c}\~ao de laborat\'orio}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \footnotesize
    \begin{block}{Pergunta}
      Oito curvas wireline conseguem prever \philab{}, \fzilab{},
      permeabilidade e HFU em um intervalo ainda n\~ao visto?
    \end{block}
    \begin{exampleblock}{Regra contra vazamento}
      Laborat\'orio entra como alvo $y$, nunca como entrada $X$.
      Ao prever FZI, \philab{}, $k_{\mathrm{lab}}$, RQI e HFU ficam fora.
    \end{exampleblock}

    \centering
    \scriptsize
    \begin{tabular}{lrr}
      \toprule
      Rela\c{c}\~ao & Vari\'avel & Pearson $r$ \\
      \midrule
      Perfil $\rightarrow$ \philab{} & Phi\_Neutron & 0,737 \\
      Perfil $\rightarrow$ \philab{} & \phiND{} & 0,701 \\
      Perfil $\rightarrow$ FZI & \phiND{} & 0,308 \\
      Laborat\'orio $\rightarrow$ FZI & RQI & 0,887 \\
      Laborat\'orio $\rightarrow$ FZI & $k_{\mathrm{lab}}$ & 0,709 \\

\begin{frame}[shrink=10]{Metodologia DEM/SC: da matriz mineral ao arcabou\c{c}o poroso}
  \scriptsize
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{1. Matriz e geometria}
      Fra\c{c}\~oes de calcita/dolomita do microCT formam
      $K_m$, $G_m$ e $\rho_m$ pela m\'edia Voigt--Reuss--Hill:
      \[
        M_{\mathrm{VRH}}=\frac{1}{2}
        \left(\sum_j f_jM_j+
        \left[\sum_j\frac{f_j}{M_j}\right]^{-1}\right).
      \]
      $\alpha$ \'e um par\^ametro efetivo da inclus\~ao porosa:
      $\alpha$ baixo aumenta a complac\^encia. O valor calibrado n\~ao
      identifica, por si s\'o, a geometria real dos poros.
    \end{block}
    \begin{block}{2. Differential Effective Medium}
      O DEM integra a evolu\c{c}\~ao de $K$ e $G$ \`a medida que poros
      secos ($K_i=G_i=0$) s\~ao inseridos at\'e a porosidade alvo $\phi$:
      \[
        (1-y)\frac{dK}{dy}=(K_i-K)P^*,\qquad
        (1-y)\frac{dG}{dy}=(G_i-G)Q^*.
      \]
      $P^*$ e $Q^*$ dependem de $K$, $G$ e $\alpha$ e s\~ao
      recalculados durante a integra\c{c}\~ao.
    \end{block}
    \column{0.49\textwidth}
    \begin{exampleblock}{3. Sa\'ida seca}
      \[
        V_p=\sqrt{\frac{K_{\mathrm{dry}}+\frac{4}{3}G_{\mathrm{dry}}}
                         {\rho_{\mathrm{dry}}}},
        \qquad
        V_s=\sqrt{\frac{G_{\mathrm{dry}}}{\rho_{\mathrm{dry}}}}.
      \]
      O SC resolve o meio de forma auto-consistente e funciona como
      controle; o DEM permanece a trilha principal.
    \end{exampleblock}
    \begin{exampleblock}{4. Satura\c{c}\~ao de Gassmann}
      \[
        K_{\mathrm{sat}}=K_{\mathrm{dry}}+
        \frac{(1-K_{\mathrm{dry}}/K_0)^2}{D},\quad
        G_{\mathrm{sat}}=G_{\mathrm{dry}},
      \]
      \[
        D=\frac{\phi}{K_f}+\frac{1-\phi}{K_0}
        -\frac{K_{\mathrm{dry}}}{K_0^2}.
      \]
      O fluido altera $K$ e $\rho$; $G$ permanece. A varia\c{c}\~ao de
      \Vp{} depende da competi\c{c}\~ao rigidez--densidade
      ($K_f=2{,}2$~GPa, $S_w=1$).
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{Calibra\c{c}\~ao por HFU e valida\c{c}\~ao leave-one-out}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{Calibra\c{c}\~ao inversa}
      \scriptsize
      Para cada HFU $h$ com plugs, ajustamos
      $\theta_h=(\alpha_h,s_h)$, onde
      $K_m\leftarrow s_hK_m$ e $G_m\leftarrow s_hG_m$:
      \[
        \theta_h^*=\arg\min_{\theta_h}
        \sqrt{\frac{1}{n_h}\sum_{j\in h}
        \left[V_{p,j}^{\mathrm{DEM}}(\theta_h)-V_{p,j}^{\mathrm{lab}}\right]^2}.
      \]
      Limites: $\alpha\in[0{,}15,0{,}95]$ e escala de matriz restrita.
    \end{block}
    \begin{exampleblock}{In-sample}
      \scriptsize
      Os mesmos plugs ajustam e avaliam o modelo. Mede fechamento da
      calibra\c{c}\~ao, mas \'e otimista: MAPE 9,1\%.
    \end{exampleblock}
    \column{0.49\textwidth}
    \begin{block}{LOO: 10 rodadas}
      \scriptsize
      Em cada rodada:
      \begin{enumerate}
        \item reservar um plug $j$;
        \item recalibrar $\theta_h^{(-j)}$ sem usar $V_{p,j}^{\mathrm{lab}}$;
        \item prever o plug reservado;
        \item reunir as dez previs\~oes de teste.
      \end{enumerate}
      \[
        \widehat{V}_{p,j}^{\mathrm{LOO}}
        =F_{\mathrm{DEM}}\!\left(x_j;\theta_h^{(-j)}\right).
      \]
    \end{block}
    \begin{exampleblock}{O que o LOO mede}
      \scriptsize
      Generaliza\c{c}\~ao entre os dez plugs do po\c{c}o~861:
      MAPE 14,8\%. N\~ao \'e valida\c{c}\~ao interpo\c{c}os.
      HFU4, sem plug, continua usando \emph{fallback}.
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{Transi\c{c}\~ao 2 $\rightarrow$ 3: corrigir o res\'iduo sem substituir a f\'isica}
  \centering
  \begin{tikzpicture}[

    \node[box,fill=orange!12,draw=orange!80!black] (m) at (0,-0.9)
      {\textbf{ML residual}\\$\widehat{\dVp}$};
    \node[box,fill=drpyellow!25,draw=orange!80!black] (h) at (6.0,0)
      {\textbf{H\'ibrido}\\
      $V_{\mathrm{p,hib}}=V_{\mathrm{p,Gassmann}}+\widehat{\dVp}$};
    \draw[arr] (f.east) -| ([xshift=-0.8cm]h.north west) -- (h.west);

    Alvo:
    $\dVp=\Vpson-V_{\mathrm{p,Gassmann}}$.
    O s\^onico entra somente em $y$. \phiS{}, DTCO, DTSM, \Vp{}/\Vs{}
    s\^onicos e o res\'iduo observado ficam fora de $X$.
  \end{exampleblock}
\end{frame}

\begin{frame}{Treinamento e valida\c{c}\~ao do ML: OOF por blocos de profundidade}
  \scriptsize
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \begin{block}{Divis\~ao espacial}
      Ordenamos as 87 linhas por profundidade e formamos tr\^es blocos
      cont\'iguos de aproximadamente 29 linhas:
      \[
        B_0=\text{topo},\qquad B_1=\text{meio},\qquad B_2=\text{base}.
      \]
      Cada rodada usa 58 linhas para treino e 29 para teste.
    \end{block}
    \centering
    \setlength{\tabcolsep}{3pt}
    \begin{tabular}{cll}
      \toprule
      Fold & Treino & Teste \\
      \midrule
      0 & $B_1\cup B_2$ & $B_0$ \\
      1 & $B_0\cup B_2$ & $B_1$ \\
      2 & $B_0\cup B_1$ & $B_2$ \\
      \bottomrule
    \end{tabular}
    \vspace{0.08cm}
    \raggedright
    O modelo, o tratamento das vari\'aveis e qualquer estat\'istica de
    treino s\~ao ajustados apenas nos dois blocos de treinamento.
    \column{0.46\textwidth}
    \begin{exampleblock}{Alvo e predi\c{c}\~ao}
      \[
        \Delta V_{p,i}=V_{p,i}^{\mathrm{sonico}}
        -V_{p,i}^{\mathrm{Gassmann}},
      \]
      \[
        \widehat{\Delta V}_{p,i}
        =\widehat{g}^{(-k)}(X_i),\quad i\in B_k.
      \]
      O h\'ibrido de teste \'e
      $V_{p,i}^{\mathrm{hib}}=V_{p,i}^{\mathrm{Gassmann}}
      +\widehat{\Delta V}_{p,i}$.
    \end{exampleblock}
    \begin{block}{OOF final}
      \scriptsize
      Concatenamos as predi\c{c}\~oes dos tr\^es blocos. Cada profundidade
      aparece uma vez no teste e nunca foi vista pelo seu modelo:
      \[
        \widehat{\Delta V}_{p}^{\mathrm{OOF}}
        =[\widehat{\Delta V}_{p,B_0};
          \widehat{\Delta V}_{p,B_1};
          \widehat{\Delta V}_{p,B_2}].
      \]
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Por que OOF por profundidade, e n\~ao divis\~ao aleat\'oria}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{Depend\^encia espacial}
      \scriptsize
      Linhas vizinhas compartilham f\'acies, porosidade e resposta
      el\'astica. Um sorteio aleat\'orio coloca vizinhos quase id\^enticos
      no treino e no teste, produzindo uma estimativa otimista.
    \end{block}
    \begin{block}{O que evitamos}
      \scriptsize
      \begin{itemize}
        \item s\^onico, DTCO/DTSM e \phiS{} em $X$;
        \item res\'iduo observado como entrada;
        \item ajuste de hiperpar\^ametros no bloco de teste;
        \item recalibra\c{c}\~ao DEM/SC usando o DSI.
      \end{itemize}
    \end{block}
    \column{0.49\textwidth}
    \begin{exampleblock}{O que a OOF avalia}
      \scriptsize
      Previsibilidade do \emph{corretor residual} em trechos reservados
      do pr\'oprio po\c{c}o~861. As 87 predi\c{c}\~oes de teste podem ser
      confrontadas ao DSI sem usar previs\~oes de treino.
    \end{exampleblock}
    \begin{exampleblock}{O que a OOF n\~ao demonstra}
      \scriptsize
      N\~ao valida a cadeia completa de forma independente: a f\'isica
      foi calibrada nos plugs do mesmo intervalo e os r\'otulos de treino
      usam o s\^onico dos outros blocos. Transfer\^encia interpo\c{c}os
      permanece necess\'aria.
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{O que os resultados demonstram -- e quais quest\~oes permanecem abertas}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{Evid\^encia obtida}
      \scriptsize
      \begin{itemize}
        \item \phiND{} \'e uma entrada conservadora para a f\'isica.
        \item DEM/SC calibrado por HFU lab.\ fecha a ordem de grandeza
              de \Vp{}.
        \item O res\'iduo p\'os-Gassmann \'e previs\'ivel em blocos
              reservados do po\c{c}o~861.
        \item Corretor simples (inclusive linear) captura boa parte
              desse sinal.
      \end{itemize}
    \end{block}
    \column{0.49\textwidth}
    \begin{exampleblock}{Quest\~oes abertas}
      \scriptsize
      \begin{itemize}
        \item 87 linhas em apenas 28~m; f\'acies se repetem entre blocos.
        \item OOF avalia o corretor residual, n\~ao a cadeia completa
              em po\c{c}o independente.
        \item HFU4 tem seis linhas e nenhum plug; HFU do perfil exige
              r\'otulo laboratorial.
        \item O res\'iduo estat\'istico pode misturar press\~ao,
              anisotropia, fluido e escala sem identific\'a-los.
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.10cm}
  \centering
  \textbf{Consequ\^encia:} n\~ao usar o h\'ibrido em invers\~ao ou em outro
  po\c{c}o sem valida\c{c}\~ao externa.
\end{frame}

\begin{frame}{Uso operacional: qual trilha escolher}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Usar Gassmann puro}
      \scriptsize
      \begin{itemize}
        \item Po\c{c}o novo sem s\^onico denso no treino.
        \item Intervalos em que a HFU2 domina e a f\'isica j\'a fecha.
        \item Entrega que exige interpreta\c{c}\~ao estritamente f\'isica.
      \end{itemize}
    \end{block}
    \column{0.50\textwidth}
    \begin{exampleblock}{Usar h\'ibrido como trilha adicional}
      \scriptsize
      \begin{itemize}
        \item Mesmo po\c{c}o e intervalo do treinamento.
        \item Reportar Gassmann, h\'ibrido linear e RF em paralelo.
        \item Preservar a regra anti-vazamento e as predi\c{c}\~oes OOF.
        \item Declarar confian\c{c}a reduzida em HFU3/4 e a depend\^encia
              da HFU laboratorial.
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.12cm}
  \begin{block}{Pr\'oximo teste necess\'ario}
    \scriptsize
    Aplicar no po\c{c}o~1045 sem retreinar no seu s\^onico
    (transfer\^encia do corretor) e, separadamente, repetir todo o protocolo
    local no 1045 (reprodutibilidade do m\'etodo).
  \end{block}
\end{frame}

\begin{frame}{Conclus\~oes}
  \footnotesize
  \begin{enumerate}
    \item \textbf{Etapa 1:} no cen\'ario estudado, \phiND{} \'e o elo
          utiliz\'avel entre perfil e f\'isica; FZI, permeabilidade e HFU
          n\~ao foram recuper\'aveis de forma confi\'avel pelos perfis.
    \item \textbf{Etapa 2:} DEM/SC + Gassmann fornecem \Vp{} interpret\'avel
          (MAPE 7,3\%; vi\'es $+0{,}27$~km/s), condicionados \`a HFU
          laboratorial e \`a calibra\c{c}\~ao dispon\'ivel.
    \item \textbf{Etapa 3:} a corre\c{c}\~ao residual reduz o MAPE OOF para
          2,1\% (linear) e 2,8\% (RF); ganhos expressivos ocorrem mesmo
          com modelos de baixa complexidade.
    \item \textbf{Diagn\'ostico por HFU:} o ganho concentra-se onde a
          calibra\c{c}\~ao f\'isica \'e mais fraca (HFU3/4); HFU2 permanece
          est\'avel.
    \item \textbf{Alcance:} a OOF mostra previsibilidade local do res\'iduo
          no po\c{c}o~861; a transfer\^encia operacional e a quantifica\c{c}\~ao
          formal de incerteza preditiva permanecem abertas.
  \end{enumerate}
\end{frame}
