\begin{frame}{Premissas da modelagem física}
  \footnotesize
  Decisões fixadas \emph{antes} dos testes A/B --- não ajustadas ao sónico.
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{Petrofísica e textura}
      \scriptsize
      \begin{itemize}
        \item Porosidade de entrada: \phiND{} (perfil) / \philab{} (plugs).
        \item Segmentação por \textbf{HFU de laboratório}.
        \item Solido~1 $\rightarrow$ calcita; Solido~2 $\rightarrow$ dolomita
              (VRH; normalizado no CT).
        \item $\alpha$ inicial $=$ mediana microCT por HFU;
              depois calibração em $[0{,}15,\,0{,}95]$.
        \item HFU~4 sem plugs: \emph{fallback} (média HFU~2 e~3).
      \end{itemize}
    \end{block}
    \column{0.49\textwidth}
    \begin{block}{Elástica e fluido}
      \scriptsize
      \begin{itemize}
        \item DEM Berryman a seco; SC só como controle.
        \item Inclusões vazias ($K_i = G_i = 0$).
        \item Gassmann \emph{depois} do arcabouço seco
              ($K_f \approx 2{,}2$~GPa, $S_w = 1$).
        \item $K_{\mathrm{scale}}$, $G_{\mathrm{scale}}$ e $\alpha$
              calibrados no \textbf{lab}, não no DSI.
        \item Limiar A/B multiescala: ganho MAPE $\geq 2$~p.p.
              (in-sample \emph{e} LOO / perfil).
      \end{itemize}
    \end{block}
  \end{columns}
  \vspace{0.08cm}
  \begin{exampleblock}{Escopo}
    \scriptsize
    Prova de conceito \textbf{local} no poço~861 --- não implica transferência
    interpoços. ML direto perfil~$\rightarrow$\Vp{} fica fora desta etapa.
  \end{exampleblock}
\end{frame}

\begin{frame}{Ordem das inclusões: volume decrescente (literatura)}
  \footnotesize
  DEM sequencial é \emph{path-dependent}; a ordem precisa ser explícita.
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \begin{block}{Convenção adotada}
      \scriptsize
      \begin{enumerate}
        \item $\phi_{\mathrm{inc}} = f_{\mathrm{pore}}\times\phi_{\mathrm{total}}$
              (meso--macro e micro).
        \item Inclusões aplicadas em ordem de
              $\phi_{\mathrm{inc}}$ \textbf{decrescente}
              (maior volume $\rightarrow$ menor).
        \item Mesma regra da referência Equinor/Berryman do projeto.
      \end{enumerate}
    \end{block}
    \vspace{0.04cm}
    {\scriptsize
    \textbf{Não} é ``maior impacto elástico primeiro'': poros achatados
    (baixo $\alpha$) amolecem mais por unidade de volume do que poros
    arredondados. Volume $\neq$ impacto elástico.}
    \column{0.46\textwidth}
    \centering
    \begin{tikzpicture}[
      node distance=0.42cm,
      box/.style={
        draw=po861blue, thick, rounded corners=3pt, fill=po861blue!10,

        (em geral meso--macro)
      };
      \node[box, below=of a, fill=orange!12, draw=orange!70!black] (b) {%
        2.\ Inclusão de menor $\phi_{\mathrm{inc}}$\\
        (em geral micro)
      };
      \node[box, below=of b] (o) {$K_{\mathrm{dry}}$, $G_{\mathrm{dry}}$
        $\rightarrow$ Gassmann};
      \draw[arr] (m) -- (a);
      \draw[arr] (a) -- (b);
      \draw[arr] (b) -- (o);
    \end{tikzpicture}
  \end{columns}
  \vspace{0.06cm}
  \begin{block}{Boas práticas}
    \scriptsize
    Carbonatos: AR meso/macro do microCT (ou frações NMR);
    micro como inclusão crítica (abaixo da resolução ou via CMR).
    Produto monoescala quando o forward multiescala não supera o limiar.
  \end{block}
\end{frame}

\section{Protocolo}

\begin{frame}{Três escalas de validação --- três perguntas}
  \footnotesize
  Nenhuma escala sozinha aprova o modelo.
  \vspace{0.08cm}
  \centering
  \scriptsize
  \begin{tabular}{lll}
    \toprule
    Escala & Amostras & Pergunta \\
    \midrule
    Lab (linha de base) & 10 plugs CT & DEM cru vs \Vp{} lab? \\
    Lab in-sample / LOO & 10 plugs / 10 folds & Fecha o lab? Generaliza? \\
    Multiescala A/B (2f) & 10 plugs CT & DEM sequencial ganha vs M1? \\
    Perfil sónico DSI & 87 linhas & Extrapolação \emph{in situ}? \\
    Multiescala NMR (2g) & 87 linhas & Frações CMR melhoram M1? \\
    \bottomrule
  \end{tabular}
  \vspace{0.12cm}
  \begin{block}{Como ler as métricas}
    \scriptsize
    \textbf{MAPE:} erro relativo típico em \Vp{}.
    \textbf{RMSE:} penaliza outliers (km/s).
    \textbf{Viés:} sinal sistemático (positivo $=$ modelo mais rápido).
  \end{block}
\end{frame}

\section{Conclusões}

\begin{frame}{Conclusões e recomendações}
  \footnotesize
  \begin{enumerate}
    \item \textbf{Viabilidade local:} DEM/SC por HFU fecha o perfil
          ($\approx 9\%$ seco, $\approx 7\%$ com Gassmann) e o lab sob a
          métrica mais severa (LOO $\approx 15\%$).
    \item \textbf{Uso por HFU:} 3\% (HFU~2), 8\% (HFU~1), 14\% (HFU~3),
          30\% (HFU~4) --- o erro segue o número de plugs; evitar HFU~4
          sem novos dados.
    \item \textbf{Entrada:} manter \phiND{}; microCT na geometria, não no ML.
    \item \textbf{Multiescala (2f/2g):} CT forward e NMR ponto a ponto não
          superam o limiar de 2~p.p.\ --- manter monoescala Gassmann.
    \item \textbf{Gassmann:} MAPE 8,7\% $\rightarrow$ 7,3\%, viés
          $+0{,}30 \rightarrow +0{,}27$~km/s; melhora consistente,
          sem fechar o gap.
    \item \textbf{Etapa~3:} ML sobre o \emph{resíduo} (sónico $-$ física),
          com prioridade nas HFU~3 e~4 --- sem substituir a cadeia DEM/SC.
  \end{enumerate}
\end{frame}
