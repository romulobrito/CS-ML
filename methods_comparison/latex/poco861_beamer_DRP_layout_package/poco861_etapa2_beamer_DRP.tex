\begin{frame}{Premissas da modelagem física}
  \footnotesize
  Regras fixadas a priori. O sónico DSI só entra na validação.
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{Petrofísica e textura}
      \scriptsize
      \begin{itemize}
        \item Porosidade de entrada: \phiND{} (perfil) / \philab{} (plugs).
        \item Segmentação por \textbf{HFU de laboratório}.
        \item Solido~1 $\rightarrow$ calcita; Solido~2 $\rightarrow$ dolomita
              (VRH; normalizado no CT).
        \item $\alpha$: achatamento do poro ($1=$ esfera; baixo $=$ mais
              achatado e mais mole). Ajuste só em $[0{,}15,\,0{,}95]$.
        \item HFU~4 sem plugs: usa a média das HFU~2 e~3
              (não há medida elástica nessa unidade).
      \end{itemize}
    \end{block}
    \column{0.49\textwidth}
    \begin{block}{Elástica e fluido}
      \scriptsize
      \begin{itemize}
        \item DEM Berryman a seco; SC só como controle.
        \item Inclusões vazias ($K_i = G_i = 0$).
        \item Gassmann \emph{depois} do arcabouço seco
              ($K_f \approx 2{,}2$~GPa, $S_w = 1$).
        \item $\alpha^\star$ e fator $s^\star$ (comum a $K_m$, $G_m$)
              ajustados nos \textbf{plugs} de laboratório.
        \item Limiar para adotar multiescala: ganho de MAPE $\geq 2$~p.p.
              no laboratório \emph{e} no perfil.
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Porosidade total e partição NMR}
  \footnotesize
  $\phiND{}$ diz \textbf{quanto} poro existe. O NMR diz \textbf{como}
  esse volume se reparte entre famílias (meso--macro e micro).
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{1.\ $\phiND{}$ --- porosidade total}
      \scriptsize
      $\phi_{\mathrm{total}}(z)=\phiND{}(z)$ (perfil);
      nos plugs, $\philab{}$.
      Ex.: $\phiND{}=0{,}20$ $\Rightarrow$ o DEM introduz 20\% de poro.
    \end{block}
    \vspace{0.08cm}
    \begin{block}{2.\ NMR --- partição (CMFF/BFV)}
      \scriptsize
      $f_{\mathrm{meso}}=\dfrac{\mathrm{CMFF}}{\mathrm{CMFF}+\mathrm{BFV}}$,
      $f_{\mathrm{micro}}=\dfrac{\mathrm{BFV}}{\mathrm{CMFF}+\mathrm{BFV}}$,
      com $f_{\mathrm{meso}}+f_{\mathrm{micro}}=1$.
      No CT: frações GeoDict no lugar de CMFF/BFV.
    \end{block}
    \column{0.50\textwidth}
    \begin{exampleblock}{3.\ Combinação no DEM}
      \scriptsize
      $\phi_{\mathrm{meso}}=\phiND{}f_{\mathrm{meso}}$,
      $\phi_{\mathrm{micro}}=\phiND{}f_{\mathrm{micro}}$;
      a soma recupera $\phiND{}$.
      \vspace{0.06cm}
      Ex.: BFV$=0{,}06$, CMFF$=0{,}10$
      ($\phi_{\mathrm{NMR}}=0{,}16\neq 0{,}20$)
      $\Rightarrow$ $f_{\mathrm{micro}}=0{,}375$,
      $f_{\mathrm{meso}}=0{,}625$
      $\Rightarrow$ $\phi_{\mathrm{micro}}=0{,}075$,
      $\phi_{\mathrm{meso}}=0{,}125$.
      \vspace{0.06cm}
      Cada inclusão leva seu $\alpha$ (CT por HFU).
      Próximo slide: \emph{ordem} dessas inclusões.
    \end{exampleblock}
  \end{columns}
  \vspace{0.10cm}
  {\centering
  \tiny
  \textbf{Legenda.}
  CMFF $=$ FFI (fluido livre, curva CMR).
  BFV $=$ BVI (fluido ligado / irredutível, curva CMR).
  \phiND{} $=$ porosidade neutrão--densidade (perfil).
  \philab{} $=$ porosidade de laboratório (plugs).
  \par}
\end{frame}

\begin{frame}{Ordem das inclusões: volume decrescente (literatura)}
  \footnotesize
  Com $\phi_{\mathrm{inc}}=f_{\mathrm{pore}}\times\phi_{\mathrm{total}}$ já definido
  (slide anterior), o DEM sequencial é \emph{path-dependent}: a ordem
  precisa ser explícita.
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \begin{block}{Convenção adotada}
      \scriptsize
      \begin{enumerate}
        \item Calcular $\phi_{\mathrm{inc}}$ para meso--macro e micro.
        \item Aplicar inclusões em ordem de $\phi_{\mathrm{inc}}$
              \textbf{decrescente} (maior volume $\rightarrow$ menor).
        \item Mesma regra da referência Equinor/Berryman do projeto.
      \end{enumerate}
    \end{block}
    \vspace{0.06cm}
    {\scriptsize
    A ordem segue o \textbf{volume} de cada família (Equinor/Berryman).
    Poros mais achatados (baixo $\alpha$) amolecem mais por unidade de
    volume; esse efeito entra em $\alpha$, não na ordem.}
    \vspace{0.08cm}
    \begin{block}{Boas práticas}
      \scriptsize
      Carbonatos: $\alpha$ meso/macro do microCT; micro via CT ou CMR.
      No poço~861 o modelo adotado permanece monoescala (ganho abaixo do limiar).
    \end{block}
    \column{0.46\textwidth}
    \centering

\begin{frame}{Três escalas de validação --- três perguntas}
  \footnotesize
  Nenhuma escala sozinha aprova o modelo.
  \vspace{0.08cm}
  \centering
  \scriptsize
  \begin{tabular}{lll}
    \toprule
    Escala & Amostras & Pergunta \\
    \midrule

  \begin{block}{O que as métricas informam}

    Viés positivo: o modelo está mais rápido que o observado.
  \end{block}
\end{frame}

\section{Conclusões}

\begin{frame}{Conclusões e recomendações}
  \footnotesize
  \begin{enumerate}
    \item \textbf{Viabilidade local:} DEM/SC por HFU fecha o perfil
          ($\approx 9\%$ seco, $\approx 7\%$ com Gassmann) e o lab sob a
          métrica mais severa (LOO $\approx 15\%$).
    \item \textbf{Uso por HFU:} o acordo com o DSI não é uniforme.
          HFU~2 ($\alpha=0{,}18$, 4 plugs): 3\%.
          HFU~1 ($\alpha$ no máximo 0,95, 4 plugs): 8\%.
          HFU~3 (2 plugs, $\alpha$ no máximo): 14\%.
          HFU~4 (sem plug, parâmetros herdados): 30\% ---
          unidade em que novos dados mais ajudariam.
    \item \textbf{Entrada:} \phiND{} no perfil; microCT na geometria e nos
          minerais da física.
    \item \textbf{Multiescala CT e NMR:} o ganho fica abaixo de 2~p.p.\ ---
          modelo monoescala com Gassmann.
    \item \textbf{Gassmann:} MAPE 8,7\% $\rightarrow$ 7,3\%, viés
          $+0{,}30 \rightarrow +0{,}27$~km/s; reduz o afastamento e deixa
          um residual estruturado.
    \item \textbf{Etapa~3:} ML sobre o resíduo
          (sónico $-$ física), com prioridade nas HFU~3 e~4.
  \end{enumerate}
\end{frame}
