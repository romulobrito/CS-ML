\section{Modelo em janelas (CLP)}

\begin{frame}{Fluxo do modelo em janelas (CLP)}
  \footnotesize
  Não é regressão ponto a ponto. Cada janela vê 16 profundidades; o
  autoencoder reconstrói o resíduo da janela; onde elas se sobrepõem,
  toma-se a média.
  \vspace{0.12cm}
  \centering
  \begin{tikzpicture}[
    box/.style={
      draw=po861blue, thick, rounded corners=3pt, fill=po861blue!10,
      align=center, text width=2.15cm, inner sep=3.5pt, font=\tiny
    },
    ml/.style={
      draw=orange!70!black, thick, rounded corners=3pt, fill=orange!12,
      align=center, text width=2.15cm, inner sep=3.5pt, font=\tiny
    },
    outbox/.style={
      draw=po861green, thick, rounded corners=3pt, fill=po861green!12,
      align=center, text width=2.15cm, inner sep=3.5pt, font=\tiny
    },
    arr/.style={-{Stealth[length=2.2mm]}, thick, po861blue},
  ]
    \node[box] (a) at (0,0) {\textbf{1}\\Perfis $+V_{\mathrm{p,Gassmann}}$};
    \node[box] (b) at (2.7,0) {\textbf{2}\\Janelas $L=16$\\passo $1$};
    \node[ml] (c) at (5.4,0) {\textbf{3}\\Autoencoder\\do resíduo};
    \node[box] (d) at (8.1,0) {\textbf{4}\\Média nas\\sobreposições};
    \node[outbox] (e) at (10.8,0) {\textbf{5}\\$V_{\mathrm{p,hibrido}}$};
    \draw[arr] (a) -- (b);
    \draw[arr] (b) -- (c);
    \draw[arr] (c) -- (d);
    \draw[arr] (d) -- (e);
  \end{tikzpicture}
  \vspace{0.14cm}
  \begin{exampleblock}{Reconstrução em profundidade}
    \scriptsize
    Várias janelas cobrem $z$. O resíduo costurado é a média dessas
    reconstruções; depois
    $V_{\mathrm{p,hibrido}}(z)=V_{\mathrm{p,Gassmann}}(z)+\widehat{\dVp}(z)$.
    O Random Forest, por comparação, prevê $\dVp$ em cada profundidade
    isolada.
  \end{exampleblock}
\end{frame}

\begin{frame}[shrink=5]{CLP: hiperparâmetros e por que esses valores}
  \scriptsize
  Defaults de \texttt{clp\_861\_vp\_residual.py} (análogo à Etapa~1f em
  janelas; sem tuning fino em $n=87$).
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Janela e costura}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.08em}
          \item Comprimento $L=16$; passo $1$
                (cobertura densa ao longo de 87 linhas).
          \item Costura: média uniforme nas sobreposições
                (\texttt{uniform\_mean}).
        \end{itemize}
      \end{block}
      \vspace{0.06cm}
      \begin{block}{Autoencoder / CSGM}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.08em}
          \item Latente $d_z=16$; oculto $128$
          \item Épocas AE: $200$;
                iterações CSGM: $400$;
                restarts: $3$;
                $\mathrm{lr}=0{,}05$
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Prior e consistência}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.08em}
          \item Prior Ridge $h(\mathbf{u})\to\mathbf{z}$ com
                $\alpha=1$ (mesmo $\alpha$ do Ridge pontual).
          \item Grade de $\lambda$ de consistência no protocolo
                esparso (seleção por RMSE em validação).
        \end{itemize}
      \end{block}
      \vspace{0.06cm}
      \begin{exampleblock}{Por quê}
        \scriptsize
        $L=16$ e AE $(16,128)$ repetem a escala da Etapa~1f:
        comparar CLP vs.\ RF no \emph{mesmo} espírito de janela,
        não otimizar CLP até empatar o RF.
        Com sônico denso, o RF pontual ($200$ árvores) permanece o
        corretor oficial.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Quanta calibração esparsa o modelo em janelas exige}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
    \footnotesize
    $\rho$ é a fração das profundidades de \textbf{teste} em que o resíduo
    $\dVp$ é conhecido:
    \[
      \rho = \frac{n_{\mathrm{cal,teste}}}{n_{\mathrm{teste}}}.
    \]
    $\rho=0$ é o caso operacional: poço novo, nenhum rótulo residual.
    \par
    \vspace{0.06cm}
    {\centering
    \scriptsize
    \setlength{\tabcolsep}{5pt}
    \begin{tabular}{cccc}
      \toprule
      $\rho$ & $n_{\mathrm{cal,test}}$ & CLP & RF esparso \\
      \midrule
      0 & 0 & 6,8 & 7,3 \\
      0,115 (plugs) & 3 & 6,5 & 7,5 \\
      0,30 & 9 & 4,7 & 6,2 \\
      0,50 & 14 & 4,0 & 3,7 \\
      0,70 & 20 & \textbf{2,5} & 3,8 \\
      1,00 & 29 & 1,0 & 3,5 \\
      \midrule
      \multicolumn{2}{l}{Random Forest denso} & \multicolumn{2}{c}{2,9} \\
      \bottomrule
    \end{tabular}
    \par}
    \vspace{0.08cm}
    {\scriptsize Com dez plugs ($\rho \approx 0{,}12$) as duas abordagens
    esparsas mal se distinguem do próprio Gassmann.}
    \column{0.52\textwidth}
    \slidefigure[0.80]{fig3_rho_sweep_mape}{MAPE versus $\rho$: CLP e
    Random Forest com calibração esparsa, contra Random Forest denso e Gassmann.}
  \end{columns}
\end{frame}

\section{Limites e decisão}

\begin{frame}[shrink=5]{Limites destes números}
  \footnotesize
  \vspace{0.04cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{Amostragem}
      \scriptsize
      \begin{itemize}
        \item 87 profundidades e oito entradas; Random Forest mantido
              modesto ($200$ árvores, semente $42$; defaults sklearn no
              restante) para conter sobreajuste --- escolha herdada da
              Etapa~1, não resultado de grid search.
        \item HFU~4 com $n=6$ é estatisticamente frágil --- e é justamente ela
              que concentra o maior ganho reportado.
      \end{itemize}
    \end{block}
    \begin{block}{Validação}
      \scriptsize
      \begin{itemize}
        \item O sônico entra no alvo: validação fora da amostra
              neste poço não é poço cego.
        \item 28~m com facies que se repetem entre blocos: interpolação
              em profundidade.
      \end{itemize}
    \end{block}
    \column{0.49\textwidth}
    \begin{exampleblock}{Fontes de incerteza no residual}
      \scriptsize
      \begin{itemize}
        \item Fluido: $K_f=2{,}2$~GPa e $S_w=1$ no produto; mediana de
              poço e VRH+\texttt{SWIRR} testadas --- sem ganho vs.\ sônico
              (Entrega~2 / relatório residual).
        \item Pressão efetiva, limite de Gassmann, HFU, anisotropia e
              condição de medida do sônico podem estar no mesmo desvio.
        \item Não substitui um modelo físico nem aponta um único mecanismo.
      \end{itemize}
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{Quando usar só Gassmann versus o híbrido no poço~861}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Só física (Gassmann)}
      \scriptsize
      \begin{itemize}
        \item HFU~2 (já $\approx 2\%$ MAPE).
        \item Intervalos sem perfil sônico denso e sem rótulo residual.
        \item Quando se exige interpretabilidade pura do DEM/SC.
      \end{itemize}
    \end{block}
    \column{0.50\textwidth}
    \begin{exampleblock}{Híbrido RF}
      \scriptsize
      \begin{itemize}
        \item Mesmo poço e intervalo, com sônico no alvo.
        \item HFU~3/4: correção empírica local.
        \item Reportar também a trilha linear, mais fácil de auditar.
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.10cm}
  {\scriptsize Antes de inversão sísmica: os dois testes de generalização
  abaixo --- não a validação intra-poço do 861.}
\end{frame}

\begin{frame}{Conclusões}
  \footnotesize
  \begin{enumerate}
    \item \textbf{Viabilidade:} MAPE de \Vp{} cai de 7,3\% para 2,8\% e o viés
          de $+0{,}27$ para $+0{,}01$~km/s fora da amostra neste poço.
    \item \textbf{Papel:} ML é corretor empírico da física --- não substitui o
          sônico nem é atalho perfil $\rightarrow$ \Vp{}.
    \item \textbf{HFU:} ganho em HFU~4 e~3; HFU~2 estável.
    \item \textbf{Regressor:} RF de referência (2,8\%); linear ainda melhor (2,1\%).
    \item \textbf{CLP:} 6,3--6,7\%; só supera o Random Forest denso com
          $\rho \gtrsim 0{,}7$.
  \end{enumerate}
\end{frame}
