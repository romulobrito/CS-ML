\begin{frame}[shrink=5]{Justificativa física e formulação estatística}
  \footnotesize
  Não há fórmula de Gassmann que forneça \(\dVp\) a partir dos perfis.
  O teste é empírico: além da média, o residual ainda depende de $X$?
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
    \begin{block}{Justificativa física (Etapa~2)}
      \scriptsize
      \begin{itemize}
        \item PVT fixo ($K_f=2{,}2$~GPa, $S_w=1$) e calibração DEM/SC
              por HFU (fraca onde faltam plugs).
        \item Gassmann é o limite de baixa frequência; o perfil sônico
              mede outra escala.
        \item Isso produz um viés \textbf{sistemático}
              (\(+0{,}27\)~km/s) e levanta a hipótese de que parte do
              erro tenha estrutura nos perfis --- a previsibilidade
              só se demonstra nos blocos de teste.
      \end{itemize}
    \end{block}
    \column{0.52\textwidth}
    \begin{exampleblock}{Formulação estatística}
      \scriptsize
      Decomposição do residual
      ($\mu_{\Delta}\approx -0{,}27$~km/s):
      \[
        \dVp = \mu_{\Delta} + h(X) + \varepsilon,
      \]
      \(\mu_{\Delta}\): deslocamento médio;
      \(h(X)\): variação previsível em torno da média;
      \(\varepsilon\): não previsível com $X$.
      Equivalente a reunir \(g(X)=\mu_{\Delta}+h(X)\):
      \[
        \Vpsonsub
        = V_{\mathrm{p,Gassmann}} + g(X) + \varepsilon,
      \]
      com \(g(X)\approx\mathbb{E}[\dVp\mid X]\)
      (o RF aproxima \(\widehat{g}(X)\)).
      Hipótese nula: \(H_0:\ h\equiv 0\)
      (só a média; sem estrutura em $X$).
      Preditor nulo:
      \(\widehat{\dVp}=\mu_{\Delta,\mathrm{treino}}
      \Rightarrow R^{2}_{\mathrm{teste}}\approx 0\).
      Este conjunto testa se \(h(X)\) é detectável ---
      não o mecanismo físico de \(h\).
      (No slide seguinte: decomposição física de \(g\) via Taylor.)
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}[shrink=5]
  \frametitle{Isolando o efeito da porosidade}
  \scriptsize
  \setlength{\abovedisplayskip}{1.5pt}
  \setlength{\belowdisplayskip}{1.5pt}
  \setlength{\abovedisplayshortskip}{0.5pt}
  \setlength{\belowdisplayshortskip}{0.5pt}
  Modelo completo: $V_p=F(\phi,\theta,h)$
  ($\phi$: porosidade; $\theta$: mineralogia, $\alpha$, $S_w$, $K_f$,
  pressão,\ldots; $h$: HFU).
  Fixamos HFU e $\theta$ para a parcial
  $F_\phi=\partial F/\partial\phi$:
  na mesma HFU e mesmos parâmetros, o que ocorre se
  trocarmos só \phiND{} por $\phi^{*}$ menor?
  \vspace{0.03cm}

  $\delta\phi=\phiND{}-\phi^{*}$,
  $\dVp=\Vpsonsub-F(\phiND{},\theta,h)$.
  Com $\Vpsonsub=F(\phi^{*},\theta,h)+r_{\mathrm{outros}}+\varepsilon$:
  \[
    \dVp = F(\phi^{*},\theta,h)-F(\phiND{},\theta,h)
    +r_{\mathrm{outros}}+\varepsilon.
  \]
  Taylor em \phiND{} (incremento $-\delta\phi$),
  desprezando $O(\delta\phi^{2})$:
  \[
    F(\phi^{*},\theta,h)
    \approx F(\phiND{},\theta,h)
    -F_\phi(\phiND{},\theta,h)\,\delta\phi.
  \]
  Cancelando $F(\phiND{},\theta,h)$:
  \[
    \dVp
    \approx
    \underbrace{-F_\phi(\phiND{},\theta,h)\,\delta\phi}
      _{\text{efeito isolado de }\phi}
    + r_{\mathrm{outros}} + \varepsilon.
  \]
  \vspace{0.04cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{block}{Valor esperado condicional}
        \scriptsize
        O residual $\dVp$ é o alvo.
        Definimos
        $g(X)\approx\mathbb{E}[\dVp\mid X]$:
        a parte de $\dVp$ \textbf{previsível}
        a partir das features $X$.
        O RF aproxima $\widehat{g}$.
        Com $\mathbb{E}[\varepsilon\mid X]=0$,
        \(\varepsilon\) sai da esperança e
        \[
          g(X)\approx
          \mathbb{E}\bigl[
            -F_\phi\,\delta\phi
            +r_{\mathrm{outros}}
            \bigm| X\bigr].
        \]
        Ou seja: $g$ \textbf{não} aprende $F_\phi$;
        aprende a contribuição previsível
        (parcela de $\phi$ + outros fatores
        correlacionados com $X$).
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{O que entra em $r_{\mathrm{outros}}$}
        \scriptsize
        Diferenças físicas além de $\phi$:
        $S_w$, $K_f$, frequência (Gassmann
        vs.\ sônico), pressão, anisotropia,
        mineralogia, geometria ($\alpha$),
        HFU mal atribuída, heterogeneidade
        abaixo da resolução do perfil,
        interações entre esses mecanismos.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Dedução do sinal via Taylor}
  \scriptsize
  Da expansão (slide anterior):
  $\dVp\approx -F_\phi\,\delta\phi + r_{\mathrm{outros}}+\varepsilon$.
  Se $F_\phi<0$ e $\delta\phi\ge 0$ ($\phi^{*}\le\phiND{}$), então
  $-F_\phi\delta\phi\ge 0$:
  isoladamente, $\phi^{*}$ menor \textbf{aumenta} $V_p$ e empurra
  $\dVp$ para cima.
  Observamos $\dVp=\Vpsonsub-V_{\mathrm{p,Gassmann}}<0$
  ($\mu_\Delta\approx -0{,}27$~km/s) --- sinal \textbf{contrário}.
  \vspace{0.06cm}

  \begin{exampleblock}{Conclusão cuidadosa}
    \scriptsize
    Fixados HFU e $\theta$, trocar \phiND{} por $\phi^{*}$ menor
    tem sinal contrário ao residual observado.
    Logo: $\phi$ efetiva menor \textbf{não é causa isolada}
    do viés negativo ---
    \textbf{não} implica $\delta\phi=0$ nem \phiND{} ``correta'';
    outros mecanismos precisam de contribuição negativa dominante.
  \end{exampleblock}
  \vspace{0.05cm}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{$r_{\mathrm{outros}}$ pode dominar}
        \scriptsize
        Ex.: $-F_\phi\delta\phi=+0{,}10$ e
        $r_{\mathrm{outros}}=-0{,}40$
        $\Rightarrow$ $\dVp\approx -0{,}30$~km/s.
        Em média, com $\mu_\Delta\approx -0{,}27$ e
        $\mathbb{E}[-F_\phi\delta\phi]\ge 0$,
        necessariamente $\mathbb{E}[r_{\mathrm{outros}}]<0$.
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Próximo slide}
        \scriptsize
        Taylor é local (exige $\delta\phi$ pequeno).
        A seguir: evidência \textbf{numérica} ---
        curvas $F(\phi)$ por HFU, sem depender da 1ª ordem.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

  \scriptsize
  Fixados HFU e $\theta$ (cadeia DEM--Gassmann do poço~861).
  Pontos $=$ \phiND{} observadas em cada HFU.
  Confirma a dedução de Taylor \textbf{sem} exigir $\delta\phi$ pequeno.
  \vspace{0.02cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \slidefigure[0.92]{fig3_vp_gassmann_phi_monotonicity_by_hfu}{%
        $\phi\mapsto V_{\mathrm{p,Gassmann}}(\phi)$ por HFU;

    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Mesma conclusão, via monotonicidade}
        \scriptsize
        \[
          \phi^{*}\le\phiND{}
          \;\Rightarrow\;
          F(\phi^{*})\ge F(\phiND{})
        \]
        \[
          \Delta V_p^{(\phi)}
          =
          F(\phi^{*})-F(\phiND{})
          \ge 0.
        \]
        Observado:
        \(\dVp^{\mathrm{obs}}
        =\Vpsonsub-V_{\mathrm{p,Gassmann}}<0\),
        com $\mu_\Delta\approx -0{,}27$~km/s.
      \end{block}
      \vspace{0.04cm}
      \begin{exampleblock}{Mensagem}
        \scriptsize
        Uma $\phi$ efetiva menor \textbf{elevaria} $V_p$
        na cadeia DEM--Gassmann; o sônico está
        \textbf{abaixo} da previsão.
        Logo essa hipótese \textbf{não explica isoladamente}
        o sinal do residual.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Viabilidade}

\begin{frame}[shrink=5]
  \frametitle{Como se formam as 87 previsões de teste (OOF)}
  \scriptsize
  Na formulação anterior, $g(X)\approx\mathbb{E}[\dVp\mid X]$ e o RF
  aproxima $\widehat{g}(X)$.
  Aqui, \textbf{OOF}: três estimativas
  $\widehat{g}^{(-k)}$ ($k=0,1,2$), cada uma sem o bloco $B_k$.
  \vspace{0.06cm}

  \textbf{1.~Blocos.}
  $B_0$ topo, $B_1$ meio, $B_2$ base;
  $\lvert B_k\rvert\approx 29$;
  $B_0\cup B_1\cup B_2=\{1,\ldots,87\}$ (sem sobreposição).
  Em cada $i$: $(X_i,\Delta V_{\mathrm{p},i})$,
  $\Delta V_{\mathrm{p},i}
  =V_{\mathrm{p,sonico},i}-V_{\mathrm{p,Gassmann},i}$.
  \vspace{0.05cm}

  \textbf{2.~Rodada $k$.}
  $\widehat{g}^{(-k)}$ = versão de $\widehat{g}$ treinada \textbf{sem}
  $B_k$; prediz só em $B_k$.
  \vspace{0.03cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \centering
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{cccc}
        \toprule
        $k$ & Treino & Teste & Modelo \\
        \midrule
        0 & $B_1\cup B_2$ & $B_0$ & $\widehat{g}^{(-0)}$ \\
        1 & $B_0\cup B_2$ & $B_1$ & $\widehat{g}^{(-1)}$ \\
        2 & $B_0\cup B_1$ & $B_2$ & $\widehat{g}^{(-2)}$ \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.52\textwidth}
      \[
        \widehat{g}^{(-k)}
        =\mathrm{RF}\bigl(
          \{(X_i,\Delta V_{\mathrm{p},i}):i\notin B_k\}
        \bigr)
      \]
      \[
        \widehat{\Delta V}_{\mathrm{p},i}
        =\widehat{g}^{(-k)}(X_i),\quad i\in B_k
      \]
    \end{column}
  \end{columns}
  \vspace{0.03cm}

  \textbf{3.~Reunião.}
  $29+29+29=87$ previsões (todas fora da amostra):
  \[
    \widehat{\Delta V}_{\mathrm{p}}^{\mathrm{OOF}}
    =
    \bigl[
      \widehat{\Delta V}_{\mathrm{p},B_0};\,
      \widehat{\Delta V}_{\mathrm{p},B_1};\,
      \widehat{\Delta V}_{\mathrm{p},B_2}
    \bigr].
  \]
  \vspace{0.02cm}
  \begin{exampleblock}{}
    \scriptsize
    Cada profundidade é teste \textbf{uma} vez;
    $\widehat{\Delta V}_{\mathrm{p},i}=\widehat{g}^{(-k)}(X_i)$ vem de um
    $\widehat{g}$ que \textbf{não} treinou no bloco de $i$.
  \end{exampleblock}
\end{frame}

    \begin{exampleblock}{Agregação das métricas}
      \scriptsize
      \textbf{RMSE:} com 29 pontos em cada bloco, basta combinar os três
      RMSE já calculados:
      \[
        \mathrm{RMSE}
        = \sqrt{\frac{0{,}12^{2}+0{,}16^{2}+0{,}24^{2}}{3}}
        = 0{,}18.
      \]
      \textbf{$R^2$:} usa o vetor completo $(y,\hat{y})$ das 87 profundidades:
      \[
        R^{2}
        = 1 - \frac{\sum_{i=1}^{87}(y_i-\hat{y}_i)^{2}}
                  {\sum_{i=1}^{87}(y_i-\bar{y})^{2}}
        = 0{,}79.
      \]
    \end{exampleblock}
  \end{minipage}
\end{frame}

\begin{frame}{Conclusões}
  \footnotesize
  \begin{enumerate}\setlength{\itemsep}{0.18em}
    \item Parte do desvio entre $V_{\mathrm{p,Gassmann}}$ e $\Vpsonsub$
          acompanha os perfis ao longo da profundidade no poço~861
          ($R^2 \approx 0{,}79$ nos blocos de teste).
    \item Isso justifica seguir com o corretor residual \textbf{neste}
          intervalo; não substitui um teste em outro poço.
    \item Fluido: no intervalo, a leitura $S_w=1$ / $K_f\approx 2{,}83$~GPa
          sustenta $K_f$ fixo; o valor 2,2~GPa deixa o Gassmann um pouco
          mais otimista. Com VRH+NMR (\texttt{SWIRR}), $K_f$ cai
          (~1,40~GPa). $K_f(z)$ e mediana NMR equivalem neste
          intervalo; o MAPE piora um pouco --- não substitui o
          $K_f=2{,}2$~GPa no produto; \phiND{} permanece a porosidade
          de entrada.
  \end{enumerate}
\end{frame}
