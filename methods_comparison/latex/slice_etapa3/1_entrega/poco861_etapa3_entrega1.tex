\section{Alvo e entradas}

\begin{frame}[shrink=8]{Por que resíduo e não ML direto perfil $\rightarrow$ \Vp{}}
  \footnotesize
  Aprendemos só o desvio $\dVp$. A física continua responsável pela tendência
  de $\Vp$; o modelo estatístico corrige o que o DEM/Gassmann não captura.
  \vspace{0.10cm}
  \centering
  \begin{tikzpicture}[
    box/.style={
      draw=po861blue, thick, rounded corners=3pt, fill=po861blue!12,
      align=center, text width=3.5cm, inner sep=5pt, font=\scriptsize
    },
    mlbox/.style={
      draw=orange!70!black, thick, rounded corners=3pt, fill=orange!12,
      align=center, text width=3.5cm, inner sep=5pt, font=\scriptsize
    },
    outbox/.style={
      draw=po861green, thick, rounded corners=3pt, fill=po861green!12,
      align=center, text width=3.5cm, inner sep=5pt, font=\scriptsize
    },
    arr/.style={-{Stealth[length=2.5mm]}, thick, po861blue},
    note/.style={font=\tiny, text=drpgray, align=center}
  ]
    \node[box] (fis) at (0,0.95) {%
      \textbf{Física (Etapa~2)}\\
      $V_{\mathrm{p,Gassmann}}$\\
      explica a tendência
    };
    \node[mlbox] (ml) at (0,-0.95) {%
      \textbf{ML (Etapa~3)}\\
      $\widehat{\dVp}$\\
      explica o desvio local
    };
    \node[outbox] (out) at (5.4,0) {%
      \textbf{Híbrido}\\
      $V_{\mathrm{p,hybrid}} =
       V_{\mathrm{p,Gassmann}} + \widehat{\dVp}$
    };
    \draw[arr] (fis.east) -| ([xshift=-0.8cm]out.north west) -- (out.west);
    \draw[arr, orange!70!black] (ml.east) -| ([xshift=-0.8cm]out.south west) -- (out.west);
    \node[note] at (2.7,-2.30) {%
      Alvo de treino:
      $\dVp = \Vpsonsub - V_{\mathrm{p,Gassmann}}$.\\
      Se a física está acima do sônico, $\dVp < 0$.\\
      O sônico entra só em $y$; \textbf{nunca} nas entradas $X$.
    };
  \end{tikzpicture}
  \vspace{0.06cm}
  \begin{exampleblock}{Modelo aditivo (física fixa)}
    \scriptsize
    \[
      \Vpsonsub
      = V_{\mathrm{p,Gassmann}} + \dVp,
      \qquad
      \dVp
      = \Vpsonsub - V_{\mathrm{p,Gassmann}}.
    \]
    Aprendemos só \(\dVp\). Um regressor direto \(\mathrm{perfil}\to\Vp{}\)
    misturaria escala de medida, condição \emph{in situ} e litologia, sem
    modelo físico de referência. O teste de se \(\dVp\) tem estrutura nos
    perfis fica para o conjunto seguinte.
  \end{exampleblock}
\end{frame}

\begin{frame}[shrink=8]{Oito variáveis em $X$; \phiS{} e $\Dt$ ficam de fora}
  \footnotesize
  Sete perfis de poço e $V_{\mathrm{p,Gassmann}}$. A $\Vp$ física entra em $X$
  porque o modelo corrige \emph{em torno} dela, não no lugar dela.
  \vspace{0.06cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
    \begin{block}{Entra em $X$ (oito)}
      \scriptsize
      \begin{itemize}
        \item Raios gama (GR) e densidade.
        \item Resistividade profunda e rasa.
        \item Porosidade nêutron e \phiND{}.
        \item Litotipo (categórica).
        \item $V_{\mathrm{p,Gassmann}}$ (Etapa~2).
      \end{itemize}
    \end{block}
    \column{0.51\textwidth}
    \begin{exampleblock}{Não entra em $X$}
      \scriptsize
      \begin{itemize}
        \item $\Vpsonsub$, $\Vs$ do sônico, tempo de trânsito $\Dt$
              (compressional e cisalhante) e o resíduo observado $\dVp$.
        \item \textbf{\phiS{}}: porosidade sônica de Wyllie, calculada a
              partir de $\Dtc$. Correlação $r = 0{,}995$ com $\Dtc$ e
              $r = -0{,}993$ com $\Vpsonsub$.
        \item HFU de laboratório (já usada na calibração da Etapa~2);
              só na análise \emph{depois} do treino.
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.08cm}
  \begin{block}{Por que \phiS{} e $\Dt$ não podem entrar}
    \scriptsize
    A porosidade sônica de Wyllie é o tempo de trânsito compressional
    reescrito:
    \[
      \phiS
      = \frac{\Dtc - \Delta t_{\mathrm{ma}}}{\Delta t_{f}-\Delta t_{\mathrm{ma}}},
      \qquad
      \Vpsonsub \propto \frac{1}{\Dtc}.
    \]
    Incluir \phiS{} ou $\Dt$ em $X$ copiaria $\Vpsonsub$. O ganho aparente
    seria vazamento de informação, não previsão.
  \end{block}
\end{frame}

\begin{frame}{Corretor oficial: hiperparâmetros do Random Forest}
  \footnotesize
  Produto documentado da Etapa~3: Random Forest no alvo $\dVp$
  (mesma família da Etapa~1; demais regressores na entrega de robustez).
  \vspace{0.08cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Valores usados}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.12em}
          \item \texttt{n\_estimators}${}=200$
          \item \texttt{random\_state}${}=42$
          \item Demais: defaults do
                \texttt{sklearn.ensemble.RandomForestRegressor}
                (ex.: \texttt{max\_depth=None},
                \texttt{min\_samples\_split=2})
          \item Sem padronização de $X$ (árvores são invariantes a escala)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{exampleblock}{Por que esses valores}
        \scriptsize
        \begin{itemize}\setlength{\itemsep}{0.12em}
          \item \textbf{Herdado da Etapa~1}: comparação justa com o
                protocolo já documentado.
          \item \textbf{$n=87$ linhas}: 200 árvores é complexo o bastante
                para capturar $h(X)$, mas modesto o bastante para conter
                sobreajuste (não fizemos varredura fina de
                \texttt{max\_depth}).
          \item \textbf{Semente fixa}: reprodutibilidade do OOF e do
                artefato serializado.
          \item Alternativa auditável: regressão linear (entrega~3);
                Ridge ($\alpha=1$) como contraste penalizado.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Por que blocos de profundidade, e não sorteio aleatório}
  \footnotesize
  Amostras vizinhas no poço não são independentes: fácies, porosidade e o
  próprio erro de Gassmann variam suavemente com \(z\).
  \vspace{0.08cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{block}{Dependência espacial}
      \scriptsize
      \begin{itemize}
        \item Sortear linhas mistura vizinhos de treino e teste.
        \item O modelo ``vê'' o mesmo trecho duas vezes e o \(R^2\)
              fica otimista.
        \item Bloco contíguo: o teste é um intervalo que aquela rodada
              \textbf{não} viu.
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{exampleblock}{Critério de validação}
      \scriptsize
      Em profundidades próximas,
      \[
        \mathrm{Corr}\bigl(\dVp(z),\dVp(z+\Delta z)\bigr)
        \ \text{alta para }\Delta z\text{ pequeno.}
      \]
      Validação i.i.d.\ ignora isso. O protocolo em três blocos mede
      generalização \textbf{em profundidade} neste poço --- não
      independência entre poços.
    \end{exampleblock}
  \end{columns}
  \vspace{0.08cm}
  {\scriptsize Fluxo de uma rodada: dois blocos treinam $\rightarrow$ um
  testa $\rightarrow$ as três predições de teste cobrem as 87 linhas,
  cada profundidade uma vez.}
\end{frame}

    o modelo interpola em profundidade no mesmo poço.
  \end{exampleblock}
\end{frame}

\begin{frame}{Conclusões}
  \footnotesize
  Mantemos $V_{\mathrm{p,Gassmann}}$ como tendência física e aprendemos o
  resíduo
  $\dVp = \Vpsonsub - V_{\mathrm{p,Gassmann}}$.
  \vspace{0.08cm}
  As entradas $X$ são sete perfis não acústicos mais a própria $\Vp$ de
  Gassmann. \textbf{Não} entram em $X$: a porosidade sônica \phiS{}, o tempo
  de trânsito $\Dt$ do perfil sônico, nem $\Vp$ e $\Vs$ derivados dele ---
  isso copiaria o alvo.
  \vspace{0.08cm}
  A validação usa três blocos de profundidade neste poço. A pergunta empírica
  é se esse residual tem estrutura nos logs --- ainda sem tratar a $\Vp$
  híbrida como produto fechado.
\end{frame}
